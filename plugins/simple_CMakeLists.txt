cmake_minimum_required(VERSION 3.12)
project(mini_msp_simple_plugins)

# Установка стандартных флагов для Windows
if(WIN32)
    add_definitions(-D_WIN32_WINNT=0x0601)
    add_definitions(-DNOMINMAX)
    add_definitions(-D_CRT_SECURE_NO_WARNINGS)
    
    if(MSVC)
        set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} /W3")
        set(CMAKE_C_FLAGS_RELEASE "${CMAKE_C_FLAGS_RELEASE} /MD /O2")
    endif()
endif()

# Создаем директорию для плагинов
file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/plugins)

# Простой системный плагин
add_library(system_plugin SHARED
    ../src/simple_system_plugin.c
)

# Настройка вывода
set_target_properties(system_plugin PROPERTIES
    LIBRARY_OUTPUT_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/plugins
    RUNTIME_OUTPUT_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/plugins
    PREFIX ""
)
